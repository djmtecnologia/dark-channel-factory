"""Execution identity model and anti-PID-reuse validator for Local Watchdog."""

from __future__ import annotations

import ctypes
from ctypes import wintypes
from dataclasses import dataclass
from enum import Enum
from typing import Optional, Union

from desktop_worker.watchdog.job_object import (
    ProcessIdentity,
    Win32Api,
)


PROCESS_QUERY_LIMITED_INFORMATION = 0x1000
JOB_OBJECT_QUERY = 0x0004
ERROR_INVALID_PARAMETER = 87
ERROR_FILE_NOT_FOUND = 2
ERROR_ACCESS_DENIED = 5


class IdentityStatus(str, Enum):
    """Result of validating a running process against expected execution identity."""

    VALID = "VALID"
    PID_REUSED = "PID_REUSED"
    PROCESS_NOT_FOUND = "PROCESS_NOT_FOUND"
    NOT_IN_JOB = "NOT_IN_JOB"
    UNVERIFIABLE = "UNVERIFIABLE"


@dataclass(frozen=True)
class CanonicalExecutionIdentity:
    """Canonical 11-field execution identity tuple (Sprint 2.7 / Etapa 0.8)."""

    execution_id: str
    attempt_id: str
    job_id: str
    tenant_id: str
    channel_id: str
    worker_id: str
    fencing_token: int
    pid: int
    process_create_time: float
    job_object_name: str
    browser_user_data_dir: str

    def to_process_identity(self) -> ProcessIdentity:
        """Convert to Task 1 ProcessIdentity."""
        return ProcessIdentity(
            pid=self.pid,
            process_creation_time=self.process_create_time,
            job_object_name=self.job_object_name,
        )


@dataclass(frozen=True)
class IdentityVerificationResult:
    """Outcome of IdentityValidator.verify()."""

    status: IdentityStatus
    pid: int
    expected_create_time: float
    actual_create_time: Optional[float] = None
    in_expected_job: Optional[bool] = None
    reason: str = ""

    @property
    def is_valid(self) -> bool:
        """True only when the process is confirmed to be the exact expected execution."""
        return self.status == IdentityStatus.VALID

    @property
    def is_safe_to_target(self) -> bool:
        """True only when targeting this PID will not affect an unrelated process."""
        return self.status == IdentityStatus.VALID


class IdentityValidator:
    """Validates process identity against PID reuse using create_time and Job Object."""

    def __init__(self, win32: Optional[Win32Api] = None) -> None:
        self._win32 = win32 or Win32Api()

    def verify(
        self,
        expected: Union[CanonicalExecutionIdentity, ProcessIdentity],
        *,
        tolerance_sec: float = 0.05,
    ) -> IdentityVerificationResult:
        """Verify that `expected.pid` still belongs to the exact same process and Job Object."""
        if expected.pid <= 0:
            return IdentityVerificationResult(
                status=IdentityStatus.PROCESS_NOT_FOUND,
                pid=expected.pid,
                expected_create_time=expected.process_create_time,
                reason="Invalid or non-positive PID",
            )

        h_process = self._open_process_query(expected.pid)
        if not h_process:
            err = ctypes.get_last_error() if hasattr(ctypes, "get_last_error") else 0
            if err == ERROR_ACCESS_DENIED:
                return IdentityVerificationResult(
                    status=IdentityStatus.UNVERIFIABLE,
                    pid=expected.pid,
                    expected_create_time=expected.process_create_time,
                    reason="Access denied opening process for identity check",
                )
            return IdentityVerificationResult(
                status=IdentityStatus.PROCESS_NOT_FOUND,
                pid=expected.pid,
                expected_create_time=expected.process_create_time,
                reason=f"Process {expected.pid} not found (win32_err={err})",
            )

        try:
            try:
                actual_create_time = self._win32.get_process_creation_time(h_process)
            except Exception as exc:
                return IdentityVerificationResult(
                    status=IdentityStatus.UNVERIFIABLE,
                    pid=expected.pid,
                    expected_create_time=expected.process_create_time,
                    reason=f"Failed to query process create time: {exc}",
                )

            if abs(actual_create_time - expected.process_create_time) > tolerance_sec:
                return IdentityVerificationResult(
                    status=IdentityStatus.PID_REUSED,
                    pid=expected.pid,
                    expected_create_time=expected.process_create_time,
                    actual_create_time=actual_create_time,
                    reason=(
                        f"PID {expected.pid} reused: expected create_time="
                        f"{expected.process_create_time}, actual={actual_create_time}"
                    ),
                )

            in_job: Optional[bool] = None
            if expected.job_object_name:
                in_job = self._check_process_in_named_job(
                    h_process, expected.job_object_name
                )
                if in_job is False:
                    return IdentityVerificationResult(
                        status=IdentityStatus.NOT_IN_JOB,
                        pid=expected.pid,
                        expected_create_time=expected.process_create_time,
                        actual_create_time=actual_create_time,
                        in_expected_job=False,
                        reason=(
                            f"Process {expected.pid} is not in expected Job Object "
                            f"{expected.job_object_name}"
                        ),
                    )

            return IdentityVerificationResult(
                status=IdentityStatus.VALID,
                pid=expected.pid,
                expected_create_time=expected.process_create_time,
                actual_create_time=actual_create_time,
                in_expected_job=in_job,
                reason="Process identity verified",
            )
        finally:
            self._win32.close_handle(h_process)

    def _open_process_query(self, pid: int) -> Optional[int]:
        kernel32 = getattr(self._win32, "_kernel32", None)
        if kernel32 is None:
            kernel32 = ctypes.WinDLL("kernel32", use_last_error=True)
        handle = kernel32.OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, False, pid)
        return int(handle) if handle else None

    def _check_process_in_named_job(
        self, h_process: int, job_name: str
    ) -> Optional[bool]:
        kernel32 = getattr(self._win32, "_kernel32", None)
        if kernel32 is None:
            kernel32 = ctypes.WinDLL("kernel32", use_last_error=True)
        h_job = kernel32.OpenJobObjectW(JOB_OBJECT_QUERY, False, job_name)
        if not h_job:
            return None
        try:
            return self._win32.is_process_in_job(h_process, int(h_job))
        finally:
            self._win32.close_handle(int(h_job))

class ProcessLauncherError(RuntimeError):
    """Raised when launching an isolated upload process fails."""


@dataclass(frozen=True)
class AttemptLaunchContext:
    """Context required to launch an isolated upload attempt."""

    execution_id: str
    attempt_id: str
    job_id: str
    tenant_id: str
    channel_id: str
    worker_id: str
    fencing_token: int
    browser_user_data_dir: str
    job_object_name: Optional[str] = None

    def resolved_job_object_name(self) -> str:
        """Return explicit job_object_name or canonical Local\\DCF_UPLOAD_<attempt>_<exec>."""
        if self.job_object_name and self.job_object_name.strip():
            return self.job_object_name.strip()
        return f"Local\\DCF_UPLOAD_{self.attempt_id}_{self.execution_id}"

    def validate(self) -> None:
        """Validate all required fields before process creation."""
        required = {
            "execution_id": self.execution_id,
            "attempt_id": self.attempt_id,
            "job_id": self.job_id,
            "tenant_id": self.tenant_id,
            "channel_id": self.channel_id,
            "worker_id": self.worker_id,
            "browser_user_data_dir": self.browser_user_data_dir,
        }
        missing = [k for k, v in required.items() if not v or not str(v).strip()]
        if missing:
            raise ProcessLauncherError(
                f"Missing required AttemptLaunchContext fields: {', '.join(missing)}"
            )
        if self.fencing_token < 0:
            raise ProcessLauncherError("fencing_token must be non-negative")


@dataclass
class IsolatedUploadLaunchResult:
    """Result of launching an isolated upload subprocess."""

    identity: CanonicalExecutionIdentity
    launch_result: LaunchResult
    job_manager: JobObjectProcessManager


def build_upload_environment(
    attempt_ctx: AttemptLaunchContext,
    base_env: Optional[Mapping[str, str]] = None,
) -> Dict[str, str]:
    """Build isolated environment variables with canonical attempt markers."""
    env = dict(os.environ if base_env is None else base_env)
    job_name = attempt_ctx.resolved_job_object_name()
    env.update(
        {
            "DCF_EXECUTION_ID": attempt_ctx.execution_id,
            "DCF_ATTEMPT_ID": attempt_ctx.attempt_id,
            "DCF_JOB_ID": attempt_ctx.job_id,
            "DCF_TENANT_ID": attempt_ctx.tenant_id,
            "DCF_CHANNEL_ID": attempt_ctx.channel_id,
            "DCF_WORKER_ID": attempt_ctx.worker_id,
            "DCF_FENCING_TOKEN": str(attempt_ctx.fencing_token),
            "DCF_BROWSER_USER_DATA_DIR": attempt_ctx.browser_user_data_dir,
            "DCF_JOB_OBJECT_NAME": job_name,
        }
    )
    return env


def launch_isolated_upload(
    cmd: Union[str, Sequence[str]],
    attempt_ctx: AttemptLaunchContext,
    *,
    env: Optional[Mapping[str, str]] = None,
    cwd: Optional[str] = None,
    pre_resume_hook: Optional[Callable[[CanonicalExecutionIdentity], None]] = None,
    manager: Optional[JobObjectProcessManager] = None,
) -> IsolatedUploadLaunchResult:
    """Launch an upload subprocess inside an exclusive Windows Job Object.

    Flow:
    1. Validate attempt_ctx and build isolated environment.
    2. Create Job Object (KILL_ON_JOB_CLOSE, no breakaway).
    3. Create process with CREATE_SUSPENDED.
    4. AssignProcessToJobObject + IsProcessInJob validation.
    5. Run optional pre_resume_hook (e.g. durable persistence) before ResumeThread.
    6. ResumeThread only after all validations succeed.
    """
    attempt_ctx.validate()
    job_name = attempt_ctx.resolved_job_object_name()
    merged_env = build_upload_environment(attempt_ctx, base_env=env)
    command_line = cmd if isinstance(cmd, str) else subprocess.list2cmdline(list(cmd))

    # Temporarily apply attempt markers to os.environ for CreateProcessW inheritance
    # if no custom manager is injected, restoring immediately after launch.
    prev_env: Dict[str, Optional[str]] = {}
    for k, v in merged_env.items():
        if os.environ.get(k) != v:
            prev_env[k] = os.environ.get(k)
            os.environ[k] = v

    job_mgr = manager or JobObjectProcessManager()

    try:
        launch_res = job_mgr.launch(
            command_line=command_line,
            working_dir=cwd,
            creation_flags=CREATE_NO_WINDOW,
        )
    finally:
        for k, old_val in prev_env.items():
            if old_val is None:
                os.environ.pop(k, None)
            else:
                os.environ[k] = old_val

    if not launch_res.ok or launch_res.process is None:
        err_msg = launch_res.error.message if launch_res.error else "Unknown launch failure"
        raise ProcessLauncherError(f"Failed to launch isolated upload in Job: {err_msg}")

    proc_ident = launch_res.process.identity
    identity = CanonicalExecutionIdentity(
        execution_id=attempt_ctx.execution_id,
        attempt_id=attempt_ctx.attempt_id,
        job_id=attempt_ctx.job_id,
        tenant_id=attempt_ctx.tenant_id,
        channel_id=attempt_ctx.channel_id,
        worker_id=attempt_ctx.worker_id,
        fencing_token=attempt_ctx.fencing_token,
        pid=proc_ident.pid,
        process_create_time=proc_ident.process_creation_time,
        job_object_name=getattr(proc_ident, "job_object_name", None) or job_name,
        browser_user_data_dir=attempt_ctx.browser_user_data_dir,
    )

    if pre_resume_hook is not None:
        try:
            pre_resume_hook(identity)
        except Exception as exc:
            try:
                launch_res.process.terminate_job()
            finally:
                launch_res.process.close()
            raise ProcessLauncherError(
                f"Aborted upload launch in post-launch hook: {exc}"
            ) from exc

    return IsolatedUploadLaunchResult(
        identity=identity,
        launch_result=launch_res,
        job_manager=job_mgr,
    )

        working_dir: Optional[str] = None,
        creation_flags: int = 0,
    ) -> Optional[PROCESS_INFORMATION]:
        if creation_flags & CREATE_BREAKAWAY_FROM_JOB:
            raise ValueError("CREATE_BREAKAWAY_FROM_JOB is forbidden in creation_flags.")

        startup_info = STARTUPINFOW()
        startup_info.cb = ctypes.sizeof(STARTUPINFOW)
        proc_info = PROCESS_INFORMATION()

        cmd_buf = ctypes.create_unicode_buffer(command_line)
        flags = creation_flags | CREATE_SUSPENDED

        ok = self._kernel32.CreateProcessW(
            None,
            cmd_buf,
            None,
            None,
            False,
            flags,
            None,
            working_dir,
            ctypes.byref(startup_info),
            ctypes.byref(proc_info),
        )
        if not ok:
            return None
        return proc_info

    def assign_process_to_job_object(
        self, job_handle: int, process_handle: int
    ) -> bool:
        return bool(
            self._kernel32.AssignProcessToJobObject(
                wintypes.HANDLE(job_handle), wintypes.HANDLE(process_handle)
            )
        )

    def is_process_in_job(
        self, process_handle: int, job_handle: int
    ) -> tuple[bool, bool]:
        in_job = wintypes.BOOL(False)
        ok = self._kernel32.IsProcessInJob(
            wintypes.HANDLE(process_handle),
            wintypes.HANDLE(job_handle),
            ctypes.byref(in_job),
        )
        return bool(ok), bool(in_job.value)

    def resume_thread(self, thread_handle: int) -> int:
        return int(self._kernel32.ResumeThread(wintypes.HANDLE(thread_handle)))

    def terminate_process(self, process_handle: int, exit_code: int = 1) -> bool:
        return bool(
            self._kernel32.TerminateProcess(
                wintypes.HANDLE(process_handle), exit_code
            )
        )

    def terminate_job_object(self, job_handle: int, exit_code: int = 1) -> bool:
        return bool(
            self._kernel32.TerminateJobObject(wintypes.HANDLE(job_handle), exit_code)
        )

    def query_basic_accounting(
        self, job_handle: int
    ) -> tuple[bool, Optional[JOBOBJECT_BASIC_ACCOUNTING_INFORMATION], int]:
        info = JOBOBJECT_BASIC_ACCOUNTING_INFORMATION()
        ret_len = wintypes.DWORD(0)
        ok = bool(
            self._kernel32.QueryInformationJobObject(
                wintypes.HANDLE(job_handle),
                JOB_OBJECT_BASIC_ACCOUNTING_INFORMATION,
                ctypes.byref(info),
                ctypes.sizeof(info),
                ctypes.byref(ret_len),
            )
        )
        err = 0 if ok else self.get_last_error()
        return ok, (info if ok else None), err

    def query_process_id_list(
        self, job_handle: int, capacity: int = 64
    ) -> tuple[bool, Any, int]:
        struct_cls = _make_process_id_list_struct(capacity)
        info = struct_cls()
        ret_len = wintypes.DWORD(0)
        ok = bool(
            self._kernel32.QueryInformationJobObject(
                wintypes.HANDLE(job_handle),
                JOB_OBJECT_BASIC_PROCESS_ID_LIST,
                ctypes.byref(info),
                ctypes.sizeof(info),
                ctypes.byref(ret_len),
            )
        )
        err = 0 if ok else self.get_last_error()
        return ok, info, err

    def get_process_creation_time(self, process_handle: int) -> Optional[int]:
        creation = wintypes.FILETIME()
        exit_t = wintypes.FILETIME()
        kernel_t = wintypes.FILETIME()
        user_t = wintypes.FILETIME()
        ok = self._kernel32.GetProcessTimes(
            wintypes.HANDLE(process_handle),
            ctypes.byref(creation),
            ctypes.byref(exit_t),
            ctypes.byref(kernel_t),
            ctypes.byref(user_t),
        )
        if not ok:
            return None
        return (creation.dwHighDateTime << 32) | creation.dwLowDateTime

    def wait_for_single_object(self, handle: int, timeout_ms: int) -> int:
        return int(
            self._kernel32.WaitForSingleObject(wintypes.HANDLE(handle), timeout_ms)
        )

    def get_exit_code_process(self, process_handle: int) -> tuple[bool, int]:
        code = wintypes.DWORD(0)
        ok = bool(
            self._kernel32.GetExitCodeProcess(
                wintypes.HANDLE(process_handle), ctypes.byref(code)
            )
        )
        return ok, int(code.value)

    def close_handle(self, handle: int) -> bool:
        return bool(self._kernel32.CloseHandle(wintypes.HANDLE(handle)))

class ManagedJobProcess:
    """Represents a process launched and isolated inside a Windows Job Object."""

    def __init__(self, api: Win32Api, identity: ProcessIdentity) -> None:
        self._api = api
        self.identity = identity
        self._closed = False

    @property
    def pid(self) -> int:
        return self.identity.pid

    @property
    def state(self) -> LauncherState:
        return self.identity.state

    @property
    def is_closed(self) -> bool:
        return self._closed

    def verify_identity(self) -> bool:
        """Verify that the process handle still refers to the original process."""
        if self._closed or not self.identity.process_handle:
            return False
        if self.identity.process_creation_time is None:
            return False
        current_ct = self._api.get_process_creation_time(
            self.identity.process_handle
        )
        if current_ct is None:
            return False
        return current_ct == self.identity.process_creation_time

    def query_basic_accounting(self) -> tuple[Optional[BasicAccountingInfo], Optional[JobObjectError]]:
        if self._closed or not self.identity.job_handle:
            return None, JobObjectError(
                stage="query_basic_accounting",
                code="JOB_HANDLE_CLOSED",
                message="Job handle is closed or unavailable.",
            )
        ok, info, err = self._api.query_basic_accounting(self.identity.job_handle)
        if not ok or info is None:
            return None, JobObjectError(
                stage="query_basic_accounting",
                code="QUERY_FAILED",
                message="QueryInformationJobObject(BasicAccountingInformation) failed.",
                winerror=err,
            )
        return (
            BasicAccountingInfo(
                active_processes=int(info.ActiveProcesses),
                total_processes=int(info.TotalProcesses),
                total_terminated_processes=int(info.TotalTerminatedProcesses),
            ),
            None,
        )

    def query_process_id_list(
        self, capacity: int = 64, max_retries: int = 3
    ) -> ProcessIdListResult:
        if self._closed or not self.identity.job_handle:
            return ProcessIdListResult(
                ok=False,
                truncated=False,
                error_message="Job handle is closed or unavailable.",
            )

        current_capacity = max(1, capacity)
        attempts = 0
        last_pids: List[int] = []
        last_assigned = 0
        last_returned = 0

        while attempts <= max_retries:
            attempts += 1
            ok, info, err = self._api.query_process_id_list(
                self.identity.job_handle, capacity=current_capacity
            )
            if not ok:
                if err == ERROR_MORE_DATA:
                    current_capacity *= 2
                    continue
                return ProcessIdListResult(
                    ok=False,
                    truncated=False,
                    error_code=err,
                    error_message=f"QueryInformationJobObject failed with error {err}.",
                )

            assigned = int(info.NumberOfAssignedProcesses)
            returned = int(info.NumberOfProcessIdsInList)
            pids = [int(info.ProcessIdList[i]) for i in range(returned)]
            last_pids = pids
            last_assigned = assigned
            last_returned = returned

            if assigned > returned:
                current_capacity = max(assigned, current_capacity * 2)
                continue

            return ProcessIdListResult(
                ok=True,
                pids=pids,
                assigned_count=assigned,
                returned_count=returned,
                truncated=False,
            )

        return ProcessIdListResult(
            ok=False,
            pids=last_pids,
            assigned_count=last_assigned,
            returned_count=last_returned,
            truncated=True,
            error_code=ERROR_MORE_DATA if last_assigned == 0 else None,
            error_message=(
                f"Truncated PID list after {attempts} attempts: "
                f"assigned={last_assigned}, returned={last_returned}."
            ),
        )

    def terminate_job(
        self,
        timeout_ms: int = 5000,
        poll_interval_ms: int = 50,
        exit_code: int = 1,
    ) -> TerminationResult:
        """Terminate the entire Job Object and independently verify termination."""
        if self._closed or not self.identity.job_handle:
            return TerminationResult(
                status=TerminationStatus.INCONCLUSIVE,
                reason="Job handle is closed or unavailable.",
                error=JobObjectError(
                    stage="terminate_job",
                    code="JOB_HANDLE_CLOSED",
                    message="Cannot terminate: job handle is closed.",
                ),
            )

        try:
            ok = self._api.terminate_job_object(
                self.identity.job_handle, exit_code=exit_code
            )
            if not ok:
                err = self._api.get_last_error()
                return TerminationResult(
                    status=TerminationStatus.INCONCLUSIVE,
                    reason="TerminateJobObject failed.",
                    error=JobObjectError(
                        stage="terminate_job",
                        code="TERMINATE_JOB_FAILED",
                        message="TerminateJobObject returned False.",
                        winerror=err,
                    ),
                )

            deadline = time.monotonic() + (timeout_ms / 1000.0)
            last_acct: Optional[BasicAccountingInfo] = None
            last_pids: Optional[ProcessIdListResult] = None

            while True:
                acct, acct_err = self.query_basic_accounting()
                if acct_err is not None or acct is None:
                    return TerminationResult(
                        status=TerminationStatus.INCONCLUSIVE,
                        reason="Failed to query BasicAccountingInformation after TerminateJobObject.",
                        error=acct_err,
                    )
                last_acct = acct

                pid_res = self.query_process_id_list()
                last_pids = pid_res
                if pid_res.truncated or not pid_res.ok:
                    return TerminationResult(
                        status=TerminationStatus.INCONCLUSIVE,
                        active_processes=acct.active_processes,
                        pids=pid_res.pids,
                        truncated=pid_res.truncated,
                        reason=pid_res.error_message or "PID list query failed or truncated.",
                    )

                if acct.active_processes == 0 and len(pid_res.pids) == 0:
                    self.identity.state = LauncherState.TERMINATED
                    return TerminationResult(
                        status=TerminationStatus.TREE_TERMINATED,
                        active_processes=0,
                        pids=[],
                        truncated=False,
                        reason="Verified ActiveProcesses == 0 and empty PID list.",
                    )

                if time.monotonic() >= deadline:
                    return TerminationResult(
                        status=TerminationStatus.INCONCLUSIVE,
                        active_processes=last_acct.active_processes if last_acct else None,
                        pids=last_pids.pids if last_pids else None,
                        truncated=last_pids.truncated if last_pids else False,
                        reason="Timeout waiting for Job Object processes to terminate.",
                    )

                time.sleep(poll_interval_ms / 1000.0)
        except Exception as exc:
            return TerminationResult(
                status=TerminationStatus.INCONCLUSIVE,
                reason=f"Unexpected exception during termination: {exc}",
                error=JobObjectError(
                    stage="terminate_job",
                    code="UNEXPECTED_EXCEPTION",
                    message=str(exc),
                ),
            )

    def close(self) -> None:
        """Close all open handles safely without double-close."""
        if self._closed:
            return
        first_exc: Optional[BaseException] = None
        for attr in ("thread_handle", "process_handle", "job_handle"):
            handle = getattr(self.identity, attr, None)
            if handle:
                try:
                    self._api.close_handle(handle)
                except Exception as exc:
                    # Preserve handle reference on failure; Win32 CloseHandle
                    # failure leaves handle state indeterminate.
                    if first_exc is None:
                        first_exc = exc
                else:
                    setattr(self.identity, attr, None)
        if first_exc is not None:
            raise first_exc
        self._closed = True

    def __enter__(self) -> ManagedJobProcess:
        return self

    def __exit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        self.close()

class JobObjectProcessManager:
    """Launches and manages Windows processes inside isolated Job Objects."""

    def __init__(self, api: Optional[Win32Api] = None) -> None:
        self._api = api if api is not None else Win32Api()

    @staticmethod
    def build_limit_flags() -> int:
        """Return the enforced Job Object limit flags (no breakaway allowed)."""
        flags = JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE
        assert (flags & FORBIDDEN_BREAKAWAY_FLAGS) == 0
        return flags

    def _safe_abort(
        self,
        job_handle: Optional[int],
        process_handle: Optional[int],
        thread_handle: Optional[int],
    ) -> bool:
        """Terminate suspended process if created, confirm exit, and close all handles."""
        aborted_cleanly = True
        try:
            if process_handle:
                try:
                    terminated = self._api.terminate_process(process_handle, 1)
                    wait_timeout = 3000 if terminated else 0
                    wait_res = self._api.wait_for_single_object(
                        process_handle, wait_timeout
                    )
                    if wait_res != WAIT_OBJECT_0:
                        aborted_cleanly = False
                except Exception:
                    aborted_cleanly = False
        finally:
            for h in (thread_handle, process_handle, job_handle):
                if h:
                    try:
                        if not self._api.close_handle(h):
                            aborted_cleanly = False
                    except Exception:
                        aborted_cleanly = False
        return aborted_cleanly

    def launch(
        self,
        command_line: str,
        working_dir: Optional[str] = None,
        creation_flags: int = 0,
        process_name: Optional[str] = None,
    ) -> LaunchResult:
        """Execute the mandatory 8-step secure Job Object launch sequence."""
        if creation_flags & CREATE_BREAKAWAY_FROM_JOB:
            raise ValueError("CREATE_BREAKAWAY_FROM_JOB is forbidden in creation_flags.")

        job_handle: Optional[int] = None
        process_handle: Optional[int] = None
        thread_handle: Optional[int] = None
        stage = "create_job"

        try:
            # 1. Create Job Object
            job_handle = self._api.create_job_object(None)
            if not job_handle:
                err = self._api.get_last_error()
                return LaunchResult(
                    ok=False,
                    error=JobObjectError(
                        stage=stage,
                        code="CREATE_JOB_FAILED",
                        message="CreateJobObjectW failed.",
                        winerror=err,
                        aborted_cleanly=True,
                    ),
                )

            # 2. Configure Job Object (KILL_ON_JOB_CLOSE, no breakaway)
            stage = "configure_job"
            limit_info = JOBOBJECT_EXTENDED_LIMIT_INFORMATION()
            limit_info.BasicLimitInformation.LimitFlags = self.build_limit_flags()
            if not self._api.set_information_job_object(
                job_handle, JOB_OBJECT_EXTENDED_LIMIT_INFORMATION, limit_info
            ):
                err = self._api.get_last_error()
                clean = self._safe_abort(job_handle, None, None)
                return LaunchResult(
                    ok=False,
                    error=JobObjectError(
                        stage=stage,
                        code="SET_JOB_LIMITS_FAILED",
                        message="SetInformationJobObject failed.",
                        winerror=err,
                        aborted_cleanly=clean,
                    ),
                )

            # 3. Create process with CREATE_SUSPENDED
            stage = "create_process_suspended"
            proc_info = self._api.create_process_suspended(
                command_line=command_line,
                working_dir=working_dir,
                creation_flags=creation_flags,
            )
            if proc_info is None:
                err = self._api.get_last_error()
                clean = self._safe_abort(job_handle, None, None)
                return LaunchResult(
                    ok=False,
                    error=JobObjectError(
                        stage=stage,
                        code="CREATE_PROCESS_FAILED",
                        message="CreateProcessW(CREATE_SUSPENDED) failed.",
                        winerror=err,
                        aborted_cleanly=clean,
                    ),
                )

            process_handle = int(proc_info.hProcess)
            thread_handle = int(proc_info.hThread)
            pid = int(proc_info.dwProcessId)

            # 4. Associate process to Job Object
            stage = "assign_process_to_job"
            if not self._api.assign_process_to_job_object(
                job_handle, process_handle
            ):
                err = self._api.get_last_error()
                clean = self._safe_abort(job_handle, process_handle, thread_handle)
                return LaunchResult(
                    ok=False,
                    error=JobObjectError(
                        stage=stage,
                        code="ASSIGN_JOB_FAILED",
                        message="AssignProcessToJobObject failed; aborting without fallback.",
                        winerror=err,
                        aborted_cleanly=clean,
                    ),
                )

            # 5. Validate association using IsProcessInJob
            stage = "validate_is_process_in_job"
            ok_check, in_job = self._api.is_process_in_job(
                process_handle, job_handle
            )
            if not ok_check or not in_job:
                err = self._api.get_last_error() if not ok_check else None
                clean = self._safe_abort(job_handle, process_handle, thread_handle)
                return LaunchResult(
                    ok=False,
                    error=JobObjectError(
                        stage=stage,
                        code="PROCESS_NOT_IN_JOB",
                        message="IsProcessInJob failed or returned False.",
                        winerror=err,
                        aborted_cleanly=clean,
                    ),
                )

            # 6. Confirm process identity
            stage = "confirm_identity"
            creation_time = self._api.get_process_creation_time(process_handle)
            if creation_time is None:
                err = self._api.get_last_error()
                clean = self._safe_abort(job_handle, process_handle, thread_handle)
                return LaunchResult(
                    ok=False,
                    error=JobObjectError(
                        stage=stage,
                        code="IDENTITY_CONFIRMATION_FAILED",
                        message="GetProcessTimes failed while confirming process identity.",
                        winerror=err,
                        aborted_cleanly=clean,
                    ),
                )

            # 7. Resume thread
            stage = "resume_thread"
            suspend_count = self._api.resume_thread(thread_handle)
            if suspend_count == 0xFFFFFFFF or suspend_count < 0:
                err = self._api.get_last_error()
                clean = self._safe_abort(job_handle, process_handle, thread_handle)
                return LaunchResult(
                    ok=False,
                    error=JobObjectError(
                        stage=stage,
                        code="RESUME_THREAD_FAILED",
                        message="ResumeThread failed.",
                        winerror=err,
                        aborted_cleanly=clean,
                    ),
                )

            # 8. Return managed object
            inferred_name = process_name or os.path.basename(
                command_line.strip().split()[0].strip('"')
            )
            identity = ProcessIdentity(
                pid=pid,
                process_handle=process_handle,
                thread_handle=thread_handle,
                process_creation_time=creation_time,
                job_handle=job_handle,
                process_name=inferred_name,
                command_line=command_line,
                created_at=time.time(),
                state=LauncherState.RUNNING,
            )
            return LaunchResult(
                ok=True,
                process=ManagedJobProcess(self._api, identity),
            )

        except Exception as exc:
            clean = self._safe_abort(job_handle, process_handle, thread_handle)
            return LaunchResult(
                ok=False,
                error=JobObjectError(
                    stage=stage,
                    code="UNEXPECTED_EXCEPTION",
                    message=str(exc),
                    aborted_cleanly=clean,
                ),
            )

"""Windows integration tests for Task 2: Process Launcher + IdentityValidator."""

from __future__ import annotations

import sys
import uuid

import pytest

from desktop_worker.watchdog.identity import (
    CanonicalExecutionIdentity,
    IdentityStatus,
    IdentityValidator,
)
from desktop_worker.watchdog.job_object import TerminationStatus
from desktop_worker.watchdog.process_launcher import (
    AttemptLaunchContext,
    launch_isolated_upload,
)


pytestmark = pytest.mark.skipif(
    sys.platform != "win32", reason="Windows-only Job Object integration test"
)


def test_wi01_launch_isolated_upload_and_validate_identity() -> None:
    """WI-01: Launch real Python subprocess in Job Object, verify identity, and terminate."""
    uid = uuid.uuid4().hex[:8]
    ctx = AttemptLaunchContext(
        execution_id=f"exec-{uid}",
        attempt_id=f"att-{uid}",
        job_id=f"job-{uid}",
        tenant_id="tenant-test",
        channel_id="chan-test",
        worker_id="win-test-worker",
        fencing_token=1,
        browser_user_data_dir=f"C:/tmp/profile_{uid}",
    )

    hook_called_with: list[CanonicalExecutionIdentity] = []

    res = launch_isolated_upload(
        [sys.executable, "-c", "import time; time.sleep(30)"],
        ctx,
        pre_resume_hook=lambda ident: hook_called_with.append(ident),
    )

    try:
        assert len(hook_called_with) == 1
        assert res.identity.pid > 0
        assert res.identity.process_create_time > 0
        assert res.identity.job_object_name == ctx.resolved_job_object_name()

        validator = IdentityValidator()
        valid_check = validator.verify(res.identity)
        assert valid_check.status == IdentityStatus.VALID
        assert valid_check.is_safe_to_target is True
        assert valid_check.in_expected_job in (True, None)

        # Simulate PID reuse by altering expected process_create_time
        tampered = CanonicalExecutionIdentity(
            **{**res.identity.__dict__, "process_create_time": res.identity.process_create_time + 100.0}
        )
        reused_check = validator.verify(tampered)
        assert reused_check.status == IdentityStatus.PID_REUSED
        assert reused_check.is_safe_to_target is False

        term = res.launch_result.process.terminate_job()
        assert term.status == TerminationStatus.TREE_TERMINATED
        assert term.active_processes == 0
    finally:
        if res.launch_result.process is not None:
            res.launch_result.process.close()
